"""Task1: BERT baseline for MagnaTagATune tag prediction.

Loads annotations_final.csv and clip_info_final.csv from the raw MagnaTagATune folder,
selects the top-k most frequent tags (default 50), constructs textual inputs from
clip metadata (title, artist, album), and fine-tunes a HuggingFace Transformer
model for multi-label classification. Saves the trained model, per-epoch metrics,
F1 curve plot, and example predictions to results/task1.

Usage:
    python src/task1_bert_train.py --raw-dir data/raw/MagnaTagATune --output-dir results/task1

Requirements: transformers, datasets, torch, scikit-learn, pandas, numpy, matplotlib
"""

import argparse
import csv
from pathlib import Path
import json
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.metrics import f1_score

from transformers import (
    AutoTokenizer,
    AutoConfig,
    AutoModelForSequenceClassification,
)


def read_clip_info(magnatag_dir: Path) -> pd.DataFrame:
    path = magnatag_dir / 'clip_info_final.csv'
    df = pd.read_csv(path, sep='\t', quotechar='"', engine='python', dtype=str, keep_default_na=False)
    df['clip_id'] = df['clip_id'].astype(str).str.strip()
    return df


def read_annotations(magnatag_dir: Path) -> pd.DataFrame:
    path = magnatag_dir / 'annotations_final.csv'
    df = pd.read_csv(path, sep='\t', quotechar='"', engine='python', dtype=str, keep_default_na=False)
    df['clip_id'] = df['clip_id'].astype(str).str.strip()
    return df


def select_top_k_tags(annotations: pd.DataFrame, k: int = 50):
    text_cols = [c for c in annotations.columns if c not in {'clip_id', 'mp3_path'}]
    counts = annotations[text_cols].astype(int).sum(axis=0)
    topk = counts.sort_values(ascending=False).head(k).index.tolist()
    return topk


def prepare_dataset(df: pd.DataFrame, texts_col: str, label_cols: list, tokenizer, max_length=128):
    enc = tokenizer(list(df[texts_col].astype(str).values), padding=True, truncation=True, max_length=max_length)
    labels = df[label_cols].astype(int).values.astype(np.float32)
    dataset = {
        'input_ids': enc['input_ids'],
        'attention_mask': enc['attention_mask'],
        'labels': labels.tolist(),
    }
    return dataset


def compute_metrics(eval_pred):
    logits, labels = eval_pred
    probs = 1.0 / (1.0 + np.exp(-logits))
    preds = (probs >= 0.5).astype(int)
    micro = f1_score(labels, preds, average='micro', zero_division=0)
    macro = f1_score(labels, preds, average='macro', zero_division=0)
    return {'micro_f1': float(micro), 'macro_f1': float(macro)}


def to_pytorch_dataset(dataset_dict):
    import torch

    class _DS(torch.utils.data.Dataset):
        def __init__(self, d):
            self.input_ids = [torch.tensor(x, dtype=torch.long) for x in d['input_ids']]
            self.attention_mask = [torch.tensor(x, dtype=torch.long) for x in d['attention_mask']]
            self.labels = [torch.tensor(x, dtype=torch.float) for x in d['labels']]

        def __len__(self):
            return len(self.input_ids)

        def __getitem__(self, idx):
            return {
                'input_ids': self.input_ids[idx],
                'attention_mask': self.attention_mask[idx],
                'labels': self.labels[idx],
            }

    return _DS(dataset_dict)


def save_json(path: Path, payload):
    with open(path, 'w', encoding='utf-8') as f:
        json.dump(payload, f, indent=2)


def save_training_history(history, out_dir: Path):
    save_json(out_dir / 'training_history.json', history)
    with open(out_dir / 'training_history.csv', 'w', encoding='utf-8', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=['epoch', 'train_loss', 'val_micro_f1', 'val_macro_f1'])
        writer.writeheader()
        for row in history:
            writer.writerow(row)


def plot_f1_curves(history, out_dir: Path):
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt

    epochs = [row['epoch'] for row in history]
    micro = [row['val_micro_f1'] for row in history]
    macro = [row['val_macro_f1'] for row in history]

    fig, ax = plt.subplots(figsize=(8, 4.5))
    ax.plot(epochs, micro, marker='o', label='Validation Micro-F1')
    ax.plot(epochs, macro, marker='s', label='Validation Macro-F1')
    ax.set_title('Task 1 BERT validation F1 over epochs')
    ax.set_xlabel('Epoch')
    ax.set_ylabel('F1')
    ax.grid(True, alpha=0.3)
    ax.legend()
    fig.tight_layout()
    fig.savefig(out_dir / 'f1_curves.png', dpi=200)
    plt.close(fig)


def save_example_predictions(model, tokenizer, test_df: pd.DataFrame, top_tags, device, max_length, out_dir: Path, n_examples: int = 5):
    import torch

    sample_df = test_df.head(n_examples).copy()
    texts = list(sample_df['text'].astype(str).values)
    labels = sample_df[top_tags].astype(int).values
    enc = tokenizer(texts, padding=True, truncation=True, max_length=max_length)
    input_ids = torch.tensor(enc['input_ids'], dtype=torch.long)
    attention_mask = torch.tensor(enc['attention_mask'], dtype=torch.long)

    model.eval()
    with torch.no_grad():
        outputs = model(input_ids=input_ids.to(device), attention_mask=attention_mask.to(device))
    probs = torch.sigmoid(outputs.logits).cpu().numpy()
    preds = (probs >= 0.5).astype(int)

    examples = []
    for idx, (text, true_row, pred_row, prob_row) in enumerate(zip(texts, labels, preds, probs)):
        true_tags = [tag for tag, value in zip(top_tags, true_row) if int(value) == 1]
        predicted_tags = [tag for tag, value in zip(top_tags, pred_row) if int(value) == 1]
        top_prob_tags = [
            {'tag': tag, 'probability': round(float(prob), 4)}
            for tag, prob in sorted(zip(top_tags, prob_row), key=lambda item: item[1], reverse=True)[:5]
        ]
        examples.append({
            'sample_index': idx,
            'text': text,
            'true_tags': true_tags,
            'predicted_tags': predicted_tags,
            'top_predicted_tags': top_prob_tags,
        })

    save_json(out_dir / 'example_predictions.json', examples)


def main():
    parser = argparse.ArgumentParser(description='Train BERT multi-label classifier on MagnaTagATune metadata')
    parser.add_argument('--raw-dir', type=str, default='data/raw/MagnaTagATune', help='Path to raw/MagnaTagATune folder')
    parser.add_argument('--output-dir', type=str, default='results/task1', help='Output directory for models and metrics')
    parser.add_argument('--model-name', type=str, default='distilbert-base-uncased', help='HuggingFace model name')
    parser.add_argument('--top-k', type=int, default=50, help='Number of top tags to predict')
    parser.add_argument('--epochs', type=int, default=10)
    parser.add_argument('--batch-size', type=int, default=32)
    parser.add_argument('--max-length', type=int, default=128)
    parser.add_argument('--seed', type=int, default=42)
    args = parser.parse_args()

    magnatag_dir = Path(args.raw_dir)
    out_dir = Path(args.output_dir)
    out_dir.mkdir(parents=True, exist_ok=True)

    print('Reading MagnaTagATune metadata...')
    clip_info = read_clip_info(magnatag_dir)
    annotations = read_annotations(magnatag_dir)

    print('Selecting top-k tags')
    top_tags = select_top_k_tags(annotations, k=args.top_k)
    print(f'Top {len(top_tags)} tags selected')

    text_cols = ['title', 'artist', 'album']
    metadata = clip_info.set_index('clip_id')[text_cols + ['mp3_path']]
    tags = annotations.set_index('clip_id')[top_tags]
    data = metadata.join(tags, how='inner').reset_index()

    data['text'] = data['title'].fillna('') + '. ' + data['artist'].fillna('') + '. ' + data['album'].fillna('')

    label_array = data[top_tags].astype(int).values
    any_positive = label_array.sum(axis=1) > 0
    if any_positive.sum() < len(data):
        print(f'Filtering out {len(data) - any_positive.sum()} samples with no positive tags (optional)')
        data = data[any_positive].reset_index(drop=True)

    train_df, test_df = train_test_split(data, test_size=0.2, random_state=args.seed)
    train_df, val_df = train_test_split(train_df, test_size=0.125, random_state=args.seed)

    print(f'Dataset sizes: train={len(train_df)}, val={len(val_df)}, test={len(test_df)}')

    print('Loading tokenizer and model config')
    tokenizer = AutoTokenizer.from_pretrained(args.model_name)
    config = AutoConfig.from_pretrained(args.model_name, num_labels=len(top_tags), problem_type='multi_label_classification')
    model = AutoModelForSequenceClassification.from_pretrained(args.model_name, config=config)

    train_dataset_dict = prepare_dataset(train_df, 'text', top_tags, tokenizer, max_length=args.max_length)
    val_dataset_dict = prepare_dataset(val_df, 'text', top_tags, tokenizer, max_length=args.max_length)
    test_dataset_dict = prepare_dataset(test_df, 'text', top_tags, tokenizer, max_length=args.max_length)

    train_dataset = to_pytorch_dataset(train_dataset_dict)
    val_dataset = to_pytorch_dataset(val_dataset_dict)
    test_dataset = to_pytorch_dataset(test_dataset_dict)

    import torch
    from torch.utils.data import DataLoader
    from torch.nn import BCEWithLogitsLoss
    from torch.optim import AdamW

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model.to(device)

    train_loader = DataLoader(train_dataset, batch_size=args.batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size=args.batch_size)
    test_loader = DataLoader(test_dataset, batch_size=args.batch_size)

    optimizer = AdamW(model.parameters(), lr=5e-5)
    loss_fn = BCEWithLogitsLoss()

    best_micro = -1.0
    best_state = None
    history = []

    print('Starting training...')
    for epoch in range(1, args.epochs + 1):
        model.train()
        total_loss = 0.0
        for batch in train_loader:
            input_ids = batch['input_ids'].to(device)
            attention_mask = batch['attention_mask'].to(device)
            labels = batch['labels'].to(device)

            outputs = model(input_ids=input_ids, attention_mask=attention_mask)
            logits = outputs.logits
            loss = loss_fn(logits, labels)
            loss.backward()
            optimizer.step()
            optimizer.zero_grad()
            total_loss += loss.item() * input_ids.size(0)

        avg_loss = total_loss / len(train_loader.dataset)

        model.eval()
        all_logits = []
        all_labels = []
        with torch.no_grad():
            for batch in val_loader:
                input_ids = batch['input_ids'].to(device)
                attention_mask = batch['attention_mask'].to(device)
                labels = batch['labels'].to(device)
                outputs = model(input_ids=input_ids, attention_mask=attention_mask)
                logits = outputs.logits
                all_logits.append(logits.cpu().numpy())
                all_labels.append(labels.cpu().numpy())
        all_logits = np.vstack(all_logits)
        all_labels = np.vstack(all_labels)
        val_metrics = compute_metrics((all_logits, all_labels))
        micro = val_metrics['micro_f1']
        print(f'Epoch {epoch}: train_loss={avg_loss:.4f}, val_micro_f1={micro:.4f}, val_macro_f1={val_metrics["macro_f1"]:.4f}')
        history.append({
            'epoch': epoch,
            'train_loss': round(float(avg_loss), 6),
            'val_micro_f1': round(float(micro), 6),
            'val_macro_f1': round(float(val_metrics['macro_f1']), 6),
        })

        if micro > best_micro:
            best_micro = micro
            best_state = {k: v.cpu().clone() for k, v in model.state_dict().items()}

    if best_state is not None:
        model.load_state_dict(best_state)

    print('Evaluating on test set...')
    model.eval()
    all_logits = []
    all_labels = []
    with torch.no_grad():
        for batch in test_loader:
            input_ids = batch['input_ids'].to(device)
            attention_mask = batch['attention_mask'].to(device)
            labels = batch['labels'].to(device)
            outputs = model(input_ids=input_ids, attention_mask=attention_mask)
            logits = outputs.logits
            all_logits.append(logits.cpu().numpy())
            all_labels.append(labels.cpu().numpy())
    all_logits = np.vstack(all_logits)
    all_labels = np.vstack(all_labels)
    metrics = compute_metrics((all_logits, all_labels))

    save_json(out_dir / 'top_tags.json', top_tags)
    save_json(out_dir / 'test_metrics.json', metrics)
    save_training_history(history, out_dir)
    plot_f1_curves(history, out_dir)
    save_example_predictions(model, tokenizer, test_df, top_tags, device, args.max_length, out_dir)

    print('Saving trained model and tokenizer...')
    model.save_pretrained(out_dir / 'model')
    tokenizer.save_pretrained(out_dir / 'tokenizer')
    save_json(out_dir / 'summary.json', {
        'model_name': args.model_name,
        'epochs': args.epochs,
        'batch_size': args.batch_size,
        'max_length': args.max_length,
        'top_k_tags': args.top_k,
        'train_size': len(train_df),
        'val_size': len(val_df),
        'test_size': len(test_df),
        'best_val_micro_f1': round(float(best_micro), 6),
        'test_metrics': metrics,
        'history': history,
    })

    print('Saved task 1 artifacts to', out_dir)


if __name__ == '__main__':
    main()
